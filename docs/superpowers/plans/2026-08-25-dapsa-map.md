**Interfaces:**

- [ ] **Step 1: 검증 스크립트 작성 (실패 확인용)**

```python
# scripts/validate_poi.py
# -*- coding: utf-8 -*-
import json, sys

REQUIRED = {"id", "name", "lat", "lng", "course", "order", "desc"}

def main():
    with open("poi.json", encoding="utf-8") as f:
        data = json.load(f)
    courses = data["courses"]
    pois = data["pois"]
    errors = []
    ids = set()
    for p in pois:
        missing = REQUIRED - set(p)
        if missing:
            errors.append(f"{p.get('id','?')}: 필드 누락 {missing}")
            continue
        if p["id"] in ids:
            errors.append(f"{p['id']}: id 중복")
        ids.add(p["id"])
        if p["course"] not in courses:
            errors.append(f"{p['id']}: 미정의 course '{p['course']}'")
        if not (37.2 < p["lat"] < 37.8 and 126.8 < p["lng"] < 127.2):
            errors.append(f"{p['id']}: 좌표가 서울·수원 범위 밖 ({p['lat']}, {p['lng']})")
    for cid, c in courses.items():
        if not c.get("name") or not c.get("color", "").startswith("#"):
            errors.append(f"course {cid}: name/color 형식 오류")
    if errors:
        print("\n".join(errors)); sys.exit(1)
    print(f"OK: courses={len(courses)} pois={len(pois)}")

main()
```

"use strict";
const CONFIG = {
  csvUrl: "PASTE_PUBLISHED_CSV_URL",   // 구글 시트 "웹에 게시" CSV 주소로 교체
  formUrl: "PASTE_GOOGLE_FORM_URL",    // 구글폼 주소로 교체
  refreshMs: 5 * 60 * 1000,
  center: [37.578, 126.977],
  zoom: 13
};

const state = { poiData: null, poiById: {}, markers: {}, courseLayers: {}, map: null };

function courseColor(courseId) {
  const c = state.poiData.courses[courseId];
  return c ? c.color : "#7f8c8d";
}

function initMap() {
  state.map = L.map("map", { zoomControl: true }).setView(CONFIG.center, CONFIG.zoom);
  L.tileLayer("https://tile.openstreetmap.org/{z}/{x}/{y}.png", {
    maxZoom: 19,
    attribution: '&copy; OpenStreetMap contributors'
  }).addTo(state.map);
  document.getElementById("submitBtn").href = CONFIG.formUrl;
}

function renderPois() {
  const byCourse = {};
  for (const p of state.poiData.pois) {
    state.poiById[p.name] = p;
    const layer = state.courseLayers[p.course] ||
      (state.courseLayers[p.course] = L.layerGroup().addTo(state.map));
    const m = L.circleMarker([p.lat, p.lng], {
      radius: 8, color: "#fff", weight: 2,
      fillColor: courseColor(p.course), fillOpacity: 0.95
    }).addTo(layer);
    m.bindPopup(`<b>${p.name}</b><br>${p.desc}`);
    state.markers[p.id] = m;
    if (p.order > 0) (byCourse[p.course] = byCourse[p.course] || []).push(p);
  }
  for (const [cid, list] of Object.entries(byCourse)) {
    list.sort((a, b) => a.order - b.order);
    L.polyline(list.map(p => [p.lat, p.lng]), {
      color: courseColor(cid), weight: 3, opacity: 0.5, dashArray: "6 6"
    }).addTo(state.courseLayers[cid]);
  }
  renderLegend();
}

function renderLegend() {
  const el = document.getElementById("legend");
  el.innerHTML = Object.entries(state.poiData.courses)
    .filter(([cid]) => cid !== "etc")
    .map(([cid, c]) =>
      `<label><input type="checkbox" data-course="${cid}" checked>` +
      `<span class="dot" style="background:${c.color}"></span>${c.name}</label>`).join("");
  el.querySelectorAll("input").forEach(cb => cb.addEventListener("change", () => {
    const layer = state.courseLayers[cb.dataset.course];
    if (!layer) return;
    cb.checked ? layer.addTo(state.map) : state.map.removeLayer(layer);
  }));
}

async function boot() {
  initMap();
  state.poiData = await (await fetch("poi.json")).json();
  renderPois();
}
boot();
</script>
</body>

function showBanner(msg) {
  const el = document.getElementById("banner");
  el.textContent = msg;
  el.style.display = msg ? "block" : "none";
}

function driveThumbs(cell) {
  if (!cell) return [];
  return cell.split(/[\s,]+/).filter(Boolean).map(u => {
    const m = u.match(/[-\w]{25,}/);
    return m ? `https://drive.google.com/thumbnail?id=${m[0]}&sz=w400` : null;
  }).filter(Boolean);
}

function subCard(row) {
  const photos = driveThumbs(row["사진"])
    .map(src => `<a href="${src.replace("&sz=w400", "&sz=w1600")}" target="_blank" rel="noopener">` +
                `<img src="${src}" loading="lazy" alt="제출 사진" ` +
                `onerror="this.outerHTML='<a href=&quot;${src}&quot; target=&quot;_blank&quot;>첨부 파일 열기</a>'"></a>`)
    .join("");
  const type = row["제출 유형"] || "답사 기록";
  const badge = type === "답사 기록" ? "" : `<span class="badge">${type}</span>`;
  return `<div class="sub-card"><b>${row["팀"] || "?"}</b>${badge}` +
         `<div>${(row["내용"] || "").replace(/</g, "&lt;")}</div>${photos}</div>`;
}

function buildPoiPopup(poi) {
  const subs = state.subsByPoi[poi.name] || [];
  const head = `<b>${poi.name}</b><br>${poi.desc}`;
  if (!subs.length) return head + `<div class="sub-card">아직 올라온 자료가 없어요.</div>`;
  return head + `<div style="max-height:45vh;overflow-y:auto">${subs.map(subCard).join("")}</div>`;
}

function attachSubmissions(rows) {
  state.subsByPoi = {};
  if (state.dynamicLayer) state.map.removeLayer(state.dynamicLayer);
  state.dynamicLayer = L.layerGroup().addTo(state.map);
  const unmatched = [];
  for (const row of rows) {
    const site = (row["답사지"] || "").trim();
    const lat = parseFloat(row["위도"]), lng = parseFloat(row["경도"]);
    if (state.poiById[site]) {
      (state.subsByPoi[site] = state.subsByPoi[site] || []).push(row);
    } else if (!isNaN(lat) && !isNaN(lng)) {
      L.circleMarker([lat, lng], {
        radius: 7, color: "#fff", weight: 2,
        fillColor: courseColor("etc"), fillOpacity: 0.95
      }).addTo(state.dynamicLayer)
        .bindPopup(`<b>${row["기타 장소명"] || "직접 입력 장소"}</b>` + subCard(row));
    } else {
      unmatched.push(row);
    }
  }
  for (const p of state.poiData.pois) {
    state.markers[p.id].bindPopup(buildPoiPopup(p));
    const subs = state.subsByPoi[p.name] || [];
    state.markers[p.id].setRadius(subs.length > 0 ? Math.min(8 + subs.length, 16) : 8);
    const hasMission = subs.some(r => (r["제출 유형"] || "").startsWith("미션"));
    state.markers[p.id].setStyle({ color: hasMission ? "#f39c12" : "#fff", weight: hasMission ? 3 : 2 });
  }
  renderUnmatched(unmatched);
}

function renderUnmatched(rows) {
  let el = document.getElementById("unmatched");
  if (!el) {
    el = document.createElement("div");
    el.id = "unmatched";
    el.style.cssText = "position:fixed;bottom:12px;right:12px;z-index:1100;" +
      "background:rgba(255,255,255,.95);border-radius:8px;padding:8px 10px;" +
      "font-size:12px;max-width:60vw;max-height:30vh;overflow-y:auto;" +
      "box-shadow:0 1px 4px rgba(0,0,0,.3)";
    document.body.appendChild(el);
  }
  if (!rows.length) { el.style.display = "none"; return; }
  el.style.display = "block";
  el.innerHTML = `<b>위치 미확정 제출 ${rows.length}건</b> (운영자가 시트에서 좌표를 채우면 지도에 표시됩니다)` +
    rows.map(r => `<div class="sub-card"><b>${r["팀"] || "?"}</b> ${r["답사지"] || ""} ${r["기타 장소명"] || ""}` +
      `<div>${(r["내용"] || "").replace(/</g, "&lt;")}</div></div>`).join("");
}

function loadSubmissions() {
  const test = new URLSearchParams(location.search).get("data");
  const url = test || CONFIG.csvUrl;
  if (url.startsWith("PASTE_")) { showBanner("CONFIG.csvUrl이 아직 설정되지 않았습니다."); return; }
  Papa.parse(url, {
    download: true, header: true, skipEmptyLines: true,
    complete: res => {
      showBanner("");
      state.allRows = res.data;
      attachSubmissions(state.allRows);
    },
    error: () => showBanner("자료를 불러오지 못했습니다. 잠시 후 자동으로 다시 시도합니다.")
  });
}

function fillTeamFilter() {

  const teams = [...new Set(state.allRows.map(r => (r["팀"] || "").trim()).filter(Boolean))].sort();

    (!type || (r["제출 유형"] || "").trim() === type)));
}
document.getElementById("teamFilter").addEventListener("change", applyFilters);

2. 로컬 저장소 연결·업로드:

```bash
cd C:/apps/jnu-dapsa
git remote add origin https://github.com/<계정명>/jnu-dapsa-map.git
git push -u origin main
```

## 2. Pages 켜기 (1회)

1. 저장소 → Settings → Pages
2. Source: "Deploy from a branch", Branch: `main`, 폴더 `/ (root)` → Save
3. 1~2분 후 `https://<계정명>.github.io/jnu-dapsa-map/` 에서 지도 열림

## 3. 실제 데이터 연결 (1회)

`index.html` 상단 `CONFIG`에서 두 값을 교체하고 커밋·푸시:

- `csvUrl`: 구글 시트 "웹에 게시" CSV 주소 (구글폼-체크리스트.md 3절)
- `formUrl`: 구글폼의 "보내기 → 링크" 주소

```bash
cd C:/apps/jnu-dapsa
git add index.html
git commit -m "config: 실제 시트·폼 주소 연결"
git push
```
